Code #1. Append Object to an Array & Check Array Datatype

This problem requires two essential operations: (1) appending a JavaScript object into an existing array using the push() method, and (2) verifying if a variable is genuinely an Array datatype using Array.isArray() (since typeof [] misleadingly returns 'object').

📘 1. Detailed Comprehensive Code (With Complete Architecture & Steps) 
// Step 1: Create an initial array and a new object
let studentList = [
    { id: 101, name: "Rahul", branch: "E&TC" },
    { id: 102, name: "Sneha", branch: "Computer" }
];

let newStudent = { id: 103, name: "Ayush", branch: "IT" };

// Step 2: Append the object to the array using push()
studentList.push(newStudent);
console.log("Updated Array after push:", studentList);

// Step 3: Check if variable is an array datatype
// Note: typeof studentList returns 'object', which is ambiguous!
// Correct approach 1: Array.isArray() [Recommended ES5+]
console.log("Is studentList an Array? ->", Array.isArray(studentList)); // true
console.log("Is newStudent an Array?  ->", Array.isArray(newStudent));  // false

// Correct approach 2: instanceof operator
console.log("instanceof Array check   ->", studentList instanceof Array); // true
Technical Details & Concepts:
  • array.push(item) inserts element at the end of the array and returns new length.
  • Array.isArray(val) reliably returns boolean true if operand is an array.
  • Never use typeof val === 'array' because arrays are objects in JavaScript prototype hierarchy.
⚡ 2. Exam Short Form (Super Simple & Short — Memorize & Write in 2 Minutes)

let arr = [{ id: 1 }];
let obj = { id: 2, name: "SPPU" };

// 1. Append object to array

// 2. Check if object is an array
console.log(Array.isArray(arr)); // true
console.log(Array.isArray(obj)); // false

Code #2. Anonymous Functions & Passing Function as Argument (Callback)

An anonymous function is a function defined without a name identifier. It is stored in a variable (function expression), passed as a callback argument, or invoked immediately (IIFE).

📘 1. Detailed Comprehensive Code (With Complete Architecture & Steps) 
// 1. Anonymous Function assigned to a variable
const calculateSquare = function(num) {
    return num * num;
};
console.log("Square of 5:", calculateSquare(5)); // Output: 25

// 2. Anonymous function passed as a Callback Argument
function processNumber(val, callbackFn) {
    console.log("Processing number:", val);
    let result = callbackFn(val);
    console.log("Callback output:", result);
}

// Passing inline anonymous function
processNumber(10, function(n) {
    return n * 3; // Multiplies by 3
});

// 3. Immediately Invoked Function Expression (IIFE)
(function() {
    let secret = "Encapsulated Scope";
    console.log("IIFE Executed automatically:", secret);
})();
Technical Details & Concepts:
  • Anonymous functions are NOT hoisted; they can only be called after their definition.
  • Frequently used in event listeners: btn.addEventListener('click', function() { ... }).
  • Avoids polluting the global namespace.
⚡ 2. Exam Short Form (Super Simple & Short — Memorize & Write in 2 Minutes) 
// --- EXAM SHORT FORM (Write this in SPPU Paper) ---
// Anonymous function assigned to a variable
let add = function(a, b) {
    return a + b;
};
console.log(add(10, 20)); // 30

// Anonymous callback in setTimeout
setTimeout(function() {
    console.log("Executed anonymously!");
}, 1000);

Code #3. Pass by Reference & Addition of Two Object Values

Objects in JavaScript are passed by reference (memory address). Changes made to an object's properties inside a function directly affect the original object outside.

📘 1. Detailed Comprehensive Code (With Complete Architecture & Steps) 
// Define two objects with numeric properties
let itemA = { name: "Pen", price: 25 };
let itemB = { name: "Book", price: 75 };

// Function that accepts objects by reference and adds their values
function calculateTotalBill(obj1, obj2) {
    let total = obj1.price + obj2.price;
    
    // Demonstrating pass by reference: modifying property
    obj1.discountApplied = true;
    
    return total;
}

let bill = calculateTotalBill(itemA, itemB);
console.log("Total Bill Amount:", bill); // 100
console.log("Original Item A modified:", itemA); 
// itemA now contains discountApplied: true (proves pass by reference)
Technical Details & Concepts:
  • Primitives (Number, String, Boolean) are passed by Value (copy is made).
  • Non-Primitives (Objects, Arrays) are passed by Reference (shared pointer).
⚡ 2. Exam Short Form (Super Simple & Short — Memorize & Write in 2 Minutes) 
// --- EXAM SHORT FORM (Write this in SPPU Paper) ---

    return a.val + b.val;

Code #4. Arrays as Stacks (LIFO) and Queues (FIFO)

JavaScript arrays provide built-in methods: push() and pop() for Stacks (Last-In-First-Out), and push() and shift() for Queues (First-In-First-Out).

📘 1. Detailed Comprehensive Code (With Complete Architecture & Steps) 
// ================= STACK IMPLEMENTATION (LIFO) =================
let stack = [];
// Push elements to top of stack
stack.push("Page 1");
stack.push("Page 2");
stack.push("Page 3");
console.log("Stack state:", stack); // ["Page 1", "Page 2", "Page 3"]

// Pop element from top of stack
let popped = stack.pop();
console.log("Popped element:", popped); // "Page 3"
console.log("Stack after pop:", stack);  // ["Page 1", "Page 2"]

// ================= QUEUE IMPLEMENTATION (FIFO) =================
let queue = [];
// Enqueue elements at end
queue.push("Customer 1");
queue.push("Customer 2");
queue.push("Customer 3");
console.log("Queue state:", queue);

// Dequeue element from front using shift()
let served = queue.shift();
console.log("Served customer:", served); // "Customer 1"
console.log("Queue after dequeue:", queue); // ["Customer 2", "Customer 3"]
Technical Details & Concepts:
  • Stack: push() adds to end, pop() removes from end (LIFO).
  • Queue: push() adds to end, shift() removes from front (FIFO).
⚡ 2. Exam Short Form (Super Simple & Short — Memorize & Write in 2 Minutes) 
// --- EXAM SHORT FORM (Write this in SPPU Paper) ---
// Stack (LIFO: push + pop)

s.push(10); s.push(20);
console.log(s.pop()); // 20

// Queue (FIFO: push + shift)
let q = [];
q.push(10); q.push(20);
console.log(q.shift()); // 10

Code #5. Recursive Function in JavaScript (Factorial)

A recursive function is a function that calls itself until it reaches a terminating Base Case, preventing infinite execution and stack overflow.

📘 1. Detailed Comprehensive Code (With Complete Architecture & Steps) 
// Function to compute factorial recursively: n! = n * (n-1)!
function factorial(n) {
    // 1. Base Case: stops recursion
    if (n === 0 || n === 1) {
        return 1;
    }
    // 2. Recursive Case: function calls itself
    return n * factorial(n - 1);
}

console.log("Factorial of 5:", factorial(5)); // 5 * 4 * 3 * 2 * 1 = 120
console.log("Factorial of 0:", factorial(0)); // 1
Technical Details & Concepts:
  • Every recursive function MUST have a Base Condition.
  • Calls are stacked in the browser's execution call stack.
⚡ 2. Exam Short Form (Super Simple & Short — Memorize & Write in 2 Minutes) 
// --- EXAM SHORT FORM (Write this in SPPU Paper) ---
function fact(n) {
    if (n <= 1) return 1;    // Base condition
    return n * fact(n - 1);  // Recursive call
}
console.log(fact(5)); // 120

Code #6. String Methods Used for Regular Expressions (match, search, replace, split)

JavaScript String prototype provides four built-in methods that accept Regular Expression patterns: match(), search(), replace(), and split().

📘 1. Detailed Comprehensive Code (With Complete Architecture & Steps) 
let text = "SPPU Exam 2026 is scheduled in Nov Dec 2026";

// 1. search() -> Returns index of first match (or -1)
let index = text.search(/\d{4}/);
console.log("First 4-digit number index:", index); // 10

// 2. match() -> Returns array of matched strings
let allYears = text.match(/\d{4}/g); // 'g' for global search
console.log("All matched years:", allYears); // ["2026", "2026"]

// 3. replace() -> Replaces matched substring with replacement text
let updated = text.replace(/2026/g, "2027");
console.log("Replaced text:", updated); // "SPPU Exam 2027 is scheduled in Nov Dec 2027"

// 4. split() -> Splits string into array using regex delimiter
let words = text.split(/\s+/); // Splits by one or more whitespace
console.log("Split words array:", words);
Technical Details & Concepts:
  • search() is equivalent to indexOf() with RegEx support.
  • replace(/pattern/g, newStr) replaces all matches when global g flag is present.
  • split(/\s+/) handles multiple consecutive spaces cleanly.
⚡ 2. Exam Short Form (Super Simple & Short — Memorize & Write in 2 Minutes) 
// --- EXAM SHORT FORM (Write this in SPPU Paper) ---

console.log(str.search(/\d+/));        // 5 (index)

Code #7. Lookahead and Lookbehind Assertions in JavaScript RegEx

Lookaround assertions are zero-width matches that match characters only if they are followed by (Lookahead) or preceded by (Lookbehind) a specific pattern without including that pattern in the match result.

📘 1. Detailed Comprehensive Code (With Complete Architecture & Steps) 
let prices = "Apple: $15, Banana: 20EUR, Orange: $30";

// 1. Positive Lookahead (?=pattern): Matches X only if followed by Y
let dollarPrices = prices.match(/\d+(?=\s*EUR)/);
console.log("Price followed by EUR:", dollarPrices[0]); // "20"

// 2. Negative Lookahead (?!pattern): Matches X only if NOT followed by Y
let nonEurNumbers = prices.match(/\d+(?!EUR)/g);
console.log("Numbers not followed by EUR:", nonEurNumbers);

// 3. Positive Lookbehind (?<=pattern): Matches X only if preceded by Y
let dollarAmounts = prices.match(/(?<=\$)\d+/g);
console.log("Amounts preceded by $ sign:", dollarAmounts); // ["15", "30"]

// 4. Negative Lookbehind (?<!pattern): Matches X only if NOT preceded by Y
let rawNumbers = prices.match(/(?<!\$)\b\d+\b/g);
console.log("Numbers NOT preceded by $ sign:", rawNumbers); // ["20"]
Technical Details & Concepts:
  • Positive Lookahead: X(?=Y)
  • Negative Lookahead: X(?!Y)
  • Positive Lookbehind: (?<=Y)X (Supported in modern ES2018+)
  • Negative Lookbehind: (?
⚡ 2. Exam Short Form (Super Simple & Short — Memorize & Write in 2 Minutes)

Code #8. Reverse a String, Check Palindrome & Find Length

Demonstrates built-in string methods (length, split, reverse, join) to reverse text and check if it reads identically forwards and backwards.

📘 1. Detailed Comprehensive Code (With Complete Architecture & Steps) 
function analyzeString(str) {
    // 1. Find Length
    let len = str.length;
    console.log("String Length:", len);

    // 2. Reverse String: split into chars -> reverse array -> join back
    let reversed = str.split("").reverse().join("");
    console.log("Reversed String:", reversed);

    // 3. Check Palindrome (case-insensitive)
    let isPalindrome = str.toLowerCase() === reversed.toLowerCase();
    console.log("Is Palindrome?", isPalindrome);

    return { len, reversed, isPalindrome };
}

// Test cases
analyzeString("madam"); // Palindrome: true
analyzeString("hello"); // Palindrome: false
Technical Details & Concepts:
  • Strings are immutable in JavaScript, so we convert to array via split('') to reverse.
  • join('') joins the array elements back into a string.
⚡ 2. Exam Short Form (Super Simple & Short — Memorize & Write in 2 Minutes) 
// --- EXAM SHORT FORM (Write this in SPPU Paper) ---
let s = "radar";
let rev = s.split("").reverse().join("");

console.log("Length:", s.length);          // 5
console.log("Reversed:", rev);             // "radar"
console.log("Palindrome:", s === rev);     // true

Code #10. Create, Insert, and Append Nodes in DOM

DOM methods to dynamically construct new HTML elements in memory (createElement), create text (createTextNode), and insert them into the document tree (appendChild).

📘 1. Detailed Comprehensive Code (With Complete Architecture & Steps) 
<!DOCTYPE html>
<html>
<body>
    <ul id="myList">
        <li>Item 1</li>
    </ul>
    <button onclick="addNewItem()">Add Item</button>

    <script>
        function addNewItem() {
            // 1. Create a new <li> element
            let newLi = document.createElement("li");

            // 2. Create text content
            let text = document.createTextNode("Dynamically Added Item");

            // 3. Append text to <li>
            newLi.appendChild(text);

            // 4. Style or set attributes
            newLi.style.color = "blue";

            // 5. Append new <li> to parent <ul>
            let list = document.getElementById("myList");
            list.appendChild(newLi);
        }
    </script>
</body>
</html>
Technical Details & Concepts:
  • document.createElement('tag') creates element node.
  • parent.appendChild(child) inserts child node at end of parent's children.
  • parent.insertBefore(newNode, refNode) inserts before specific element.
⚡ 2. Exam Short Form (Super Simple & Short — Memorize & Write in 2 Minutes)

Code #11. Simple Calculator Implementation using JavaScript

A complete interactive calculator handling arithmetic operations (+, -, *, /) using DOM input elements and event handling.

📘 1. Detailed Comprehensive Code (With Complete Architecture & Steps) 
<!DOCTYPE html>
<html>
<head><title>Simple Calculator</title></head>
<body>
    <h3>Simple Arithmetic Calculator</h3>
    <input type="number" id="num1" placeholder="First Number">
    <input type="number" id="num2" placeholder="Second Number"><br><br>
    
    <button onclick="calculate('+')">+</button>
    <button onclick="calculate('-')">-</button>
    <button onclick="calculate('*')">*</button>
    <button onclick="calculate('/')">/</button>

    <h4 id="result">Result: </h4>

    <script>
        function calculate(op) {
            let n1 = parseFloat(document.getElementById('num1').value);
            let n2 = parseFloat(document.getElementById('num2').value);
            let res = 0;

            if (isNaN(n1) || isNaN(n2)) {
                alert("Please enter valid numbers");
                return;
            }

            switch(op) {
                case '+': res = n1 + n2; break;
                case '-': res = n1 - n2; break;
                case '*': res = n1 * n2; break;
                case '/': res = n2 !== 0 ? (n1 / n2) : "Cannot divide by 0"; break;
            }

            document.getElementById('result').innerText = "Result: " + res;
        }
    </script>
</body>
</html>
Technical Details & Concepts:
  • Always wrap input strings with parseFloat() or Number() to prevent string concatenation during addition.
  • Handle division by zero validation safely.
⚡ 2. Exam Short Form (Super Simple & Short — Memorize & Write in 2 Minutes) 
// --- EXAM SHORT FORM (Write this in SPPU Paper) ---
function calc(n1, n2, op) {
    if (op === '+') return n1 + n2;
    if (op === '-') return n1 - n2;
    if (op === '*') return n1 * n2;
    if (op === '/') return n2 !== 0 ? n1 / n2 : "Error";
}
console.log(calc(10, 5, '+')); // 15

Code #12. Student Registration Form Validation (Name, Mobile, Email)

Form validation script ensuring input fields are not empty, mobile number is exactly 10 digits, and email follows standard pattern.

📘 1. Detailed Comprehensive Code (With Complete Architecture & Steps) 
<!DOCTYPE html>
<html>
<body>
    <h3>Student Registration Form</h3>
    <form onsubmit="return validateForm()">
        Name: <input type="text" id="uname"><br><br>
        Mobile: <input type="text" id="mobile"><br><br>
        Email: <input type="text" id="email"><br><br>
        <input type="submit" value="Register">
    </form>

    <script>
        function validateForm() {
            let name = document.getElementById('uname').value.trim();
            let mobile = document.getElementById('mobile').value.trim();
            let email = document.getElementById('email').value.trim();

            // Name validation
            if (name === "") {
                alert("Name cannot be empty!");
                return false;
            }

            // Mobile number validation (10 digits)
            let phonePattern = /^[0-9]{10}$/;
            if (!phonePattern.test(mobile)) {
                alert("Mobile number must be exactly 10 digits!");
                return false;
            }

            // Email validation
            let emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
            if (!emailPattern.test(email)) {
                alert("Please enter a valid email address!");
                return false;
            }

            alert("Validation successful! Submitting form...");
            return true;
        }
    </script>
</body>
</html>
Technical Details & Concepts:
  • Return false in the onsubmit handler to prevent default browser form submission when validation fails.
  • Use pattern.test(input) to validate regular expressions cleanly.
⚡ 2. Exam Short Form (Super Simple & Short — Memorize & Write in 2 Minutes) 
// --- EXAM SHORT FORM (Write this in SPPU Paper) ---
function validate() {
    let name = document.getElementById("name").value;
    let mob = document.getElementById("mob").value;
    
    if (name === "") { alert("Name empty"); return false; }
    if (!/^[0-9]{10}$/.test(mob)) { alert("Invalid Mobile"); return false; }
    return true;
}
